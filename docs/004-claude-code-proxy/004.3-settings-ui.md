# 004.3 - Settings UI

**Phase 004: Claude Code Proxy** -- Sub-doc 3 of 4: Settings pages for Claude Code integration.

> See also: [004.1 Anthropic Proxy](./004.1-anthropic-proxy.md) | [004.2 API Tokens](./004.2-api-tokens.md) | [004.4 Credit Metering](./004.4-credit-metering.md)

## Overview

Two settings pages under `_auth/settings/`: API token management and default pool selection. Includes a copyable `settings.json` snippet pre-filled with the user's token for one-click Claude Code setup.

## Context & Background

Phase 4 adds Claude Code as a consumer channel. Consumers need to generate API tokens and configure their default pool. The settings UI surfaces these controls and provides a ready-to-paste `settings.json` for Claude Code configuration.

## Goals

- Token management page: generate, list, revoke tokens (max 5 active)
- Default pool selector page
- Copyable `settings.json` snippet with pre-filled token
- Server functions via service binding pattern (existing convention)

## Non-Goals

- Token usage analytics (Phase 5)
- Pool browsing/marketplace (Phase 3, separate route)
- Billing/payment settings

## 2. Server Functions

File: `apps/user-application/src/server/api-token-fns.ts`

```ts
import { createServerFn } from "@tanstack/react-start"
import { getServiceBinding } from "../lib/service-binding"

export const listTokens = createServerFn({ method: "GET" })
  .handler(async ({ context }) => {
    const api = getServiceBinding(context)
    const res = await api.fetch("/api/tokens", {
      headers: context.request.headers,
    })
    if (!res.ok) throw new Error("Failed to fetch tokens")
    return res.json()
  })

export const createToken = createServerFn({ method: "POST" })
  .validator((data: { name: string; defaultPoolId?: string }) => data)
  .handler(async ({ data, context }) => {
    const api = getServiceBinding(context)
    const res = await api.fetch("/api/tokens", {
      method: "POST",
      headers: {
        ...context.request.headers,
        "Content-Type": "application/json",
      },
      body: JSON.stringify(data),
    })
    if (!res.ok) {
      const err = await res.json()
      throw new Error(err.error ?? "Failed to create token")
    }
    return res.json()
  })

export const revokeToken = createServerFn({ method: "POST" })
  .validator((data: { id: string }) => data)
  .handler(async ({ data, context }) => {
    const api = getServiceBinding(context)
    const res = await api.fetch(`/api/tokens/${data.id}`, {
      method: "DELETE",
      headers: context.request.headers,
    })
    if (!res.ok) throw new Error("Failed to revoke token")
  })

export const updateToken = createServerFn({ method: "POST" })
  .validator((data: { id: string; name?: string; defaultPoolId?: string | null }) => data)
  .handler(async ({ data, context }) => {
    const { id, ...body } = data
    const api = getServiceBinding(context)
    const res = await api.fetch(`/api/tokens/${id}`, {
      method: "PATCH",
      headers: {
        ...context.request.headers,
        "Content-Type": "application/json",
      },
      body: JSON.stringify(body),
    })
    if (!res.ok) throw new Error("Failed to update token")
    return res.json()
  })
```

## 3. API Tokens Page

File: `apps/user-application/src/routes/_auth/settings/api-tokens.tsx`

```tsx
import { createFileRoute } from "@tanstack/react-router"
import { useState } from "react"
import { listTokens, createToken, revokeToken } from "../../../server/api-token-fns"
import { TokenTable } from "../../../components/settings/token-table"
import { GenerateTokenDialog } from "../../../components/settings/generate-token-dialog"
import { SettingsJsonSnippet } from "../../../components/settings/settings-json-snippet"
import type { ApiTokenResponse, ApiTokenCreateResponse } from "@repo/data-ops/zod-schema/api-token"

export const Route = createFileRoute("/_auth/settings/api-tokens")({
  loader: () => listTokens(),
  component: ApiTokensPage,
})

function ApiTokensPage() {
  const { data: tokens } = Route.useLoaderData()
  const [newToken, setNewToken] = useState<ApiTokenCreateResponse | null>(null)
  const [dialogOpen, setDialogOpen] = useState(false)

  const handleCreate = async (name: string, defaultPoolId?: string) => {
    const result = await createToken({ data: { name, defaultPoolId } })
    setNewToken(result)
    setDialogOpen(false)
  }

  const handleRevoke = async (id: string) => {
    await revokeToken({ data: { id } })
  }

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <div>
          <h2 className="text-lg font-semibold">API Tokens</h2>
          <p className="text-sm text-muted-foreground">
            Tokens for Claude Code integration. Max 5 active.
          </p>
        </div>
        <button
          onClick={() => setDialogOpen(true)}
          disabled={tokens.length >= 5}
          className="btn btn-primary"
        >
          Generate Token
        </button>
      </div>

      {newToken && (
        <NewTokenBanner token={newToken} onDismiss={() => setNewToken(null)} />
      )}

      <TokenTable tokens={tokens} onRevoke={handleRevoke} />

      <SettingsJsonSnippet token={newToken?.token} />

      <GenerateTokenDialog
        open={dialogOpen}
        onClose={() => setDialogOpen(false)}
        onCreate={handleCreate}
      />
    </div>
  )
}
```

### 4.2 GenerateTokenDialog

File: `apps/user-application/src/components/settings/generate-token-dialog.tsx`

```tsx
import { useState } from "react"
import { PoolSelector } from "./pool-selector"

interface GenerateTokenDialogProps {
  open: boolean
  onClose: () => void
  onCreate: (name: string, defaultPoolId?: string) => Promise<void>
}

export function GenerateTokenDialog({
  open,
  onClose,
  onCreate,
}: GenerateTokenDialogProps) {
  const [name, setName] = useState("")
  const [poolId, setPoolId] = useState<string | undefined>()
  const [loading, setLoading] = useState(false)

  if (!open) return null

  const handleSubmit = async () => {
    setLoading(true)
    try {
      await onCreate(name, poolId)
      setName("")
      setPoolId(undefined)
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50">
      <div className="bg-white dark:bg-gray-900 rounded-lg p-6 w-full max-w-md space-y-4">
        <h3 className="text-lg font-semibold">Generate API Token</h3>

        <div>
          <label className="text-sm font-medium">Name</label>
          <input
            value={name}
            onChange={(e) => setName(e.target.value)}
            placeholder="e.g. My Laptop"
            className="w-full mt-1 px-3 py-2 border rounded"
          />
        </div>

        <div>
          <label className="text-sm font-medium">Default Pool (optional)</label>
          <PoolSelector value={poolId} onChange={setPoolId} />
        </div>

        <div className="flex justify-end gap-2">
          <button onClick={onClose} className="btn btn-ghost">
            Cancel
          </button>
          <button
            onClick={handleSubmit}
            disabled={!name || loading}
            className="btn btn-primary"
          >
            {loading ? "Generating..." : "Generate"}
          </button>
        </div>
      </div>
    </div>
  )
}
```

import { useQuery } from "@tanstack/react-query"
import { listUserPools } from "../../server/pool-fns"

interface PoolSelectorProps {
  value: string | undefined
  onChange: (poolId: string | undefined) => void
}

export function PoolSelector({ value, onChange }: PoolSelectorProps) {
  const { data } = useQuery({
    queryKey: ["user-pools"],
    queryFn: () => listUserPools(),
  })

  const pools = data?.data ?? []

  return (
    <select
      value={value ?? ""}
      onChange={(e) => onChange(e.target.value || undefined)}
      className="w-full mt-1 px-3 py-2 border rounded"
    >
      <option value="">No default pool</option>
      {pools.map((p) => (
        <option key={p.id} value={p.id}>
          {p.name}
        </option>
      ))}
    </select>
  )
}

function NewTokenBanner({
  token,
  onDismiss,
}: {
  token: ApiTokenCreateResponse
  onDismiss: () => void
}) {
  const [copied, setCopied] = useState(false)

  const copyToken = async () => {
    await navigator.clipboard.writeText(token.token)
    setCopied(true)
    setTimeout(() => setCopied(false), 2000)

  return (

      <p className="text-sm font-medium text-yellow-800 dark:text-yellow-200">
        Token created. Copy it now -- it won't be shown again.

      <div className="mt-2 flex items-center gap-2">
        <code className="flex-1 bg-gray-100 dark:bg-gray-800 px-3 py-2 rounded text-xs font-mono break-all">
          {token.token}
        </code>
        <button onClick={copyToken} className="btn btn-sm">
          {copied ? "Copied" : "Copy"}
        </button>
      </div>
      <button onClick={onDismiss} className="mt-2 text-xs text-yellow-700 hover:underline">
        Dismiss

  )
}

import { useState } from "react"

interface SettingsJsonSnippetProps {
  token?: string
}

export function SettingsJsonSnippet({ token }: SettingsJsonSnippetProps) {
  const [copied, setCopied] = useState(false)

  const snippet = JSON.stringify(
    {
      env: {
        ANTHROPIC_AUTH_TOKEN: token ?? "ac_YOUR_TOKEN_HERE",
        ANTHROPIC_BASE_URL: "https://api.aigent.community/v1",
      },
    },
    null,
    2
  )

  const copySnippet = async () => {
    await navigator.clipboard.writeText(snippet)
    setCopied(true)
    setTimeout(() => setCopied(false), 2000)
  }

  return (
    <div className="space-y-2">
      <h3 className="text-sm font-medium">Claude Code Configuration</h3>
      <p className="text-xs text-muted-foreground">
        Add to <code>~/.claude/settings.json</code>:
      </p>
      <div className="relative">
        <pre className="bg-gray-100 dark:bg-gray-800 rounded-lg p-4 text-xs font-mono overflow-x-auto">
          {snippet}
        </pre>
        <button
          onClick={copySnippet}
          className="absolute top-2 right-2 btn btn-sm"
        >
          {copied ? "Copied" : "Copy"}
        </button>
      </div>
    </div>
  )
}

## 5. Default Pool Page

File: `apps/user-application/src/routes/_auth/settings/default-pool.tsx`

```tsx
import { createFileRoute } from "@tanstack/react-router"
import { listTokens, updateToken } from "../../../server/api-token-fns"
import { PoolSelector } from "../../../components/settings/pool-selector"
import { useState } from "react"
import type { ApiTokenResponse } from "@repo/data-ops/zod-schema/api-token"

export const Route = createFileRoute("/_auth/settings/default-pool")({
  loader: () => listTokens(),
  component: DefaultPoolPage,
})

function DefaultPoolPage() {
  const { data: tokens } = Route.useLoaderData()
  const activeTokens = tokens.filter((t: ApiTokenResponse) => t.isActive)

  return (
    <div className="space-y-6">
      <div>
        <h2 className="text-lg font-semibold">Default Pool for Claude Code</h2>
        <p className="text-sm text-muted-foreground">
          Set which pool each API token uses by default. Override per-request with <code>X-Pool-Id</code> header.
        </p>
      </div>

      {activeTokens.length === 0 ? (
        <p className="text-sm text-muted-foreground">
          No active tokens. Generate one in API Tokens settings first.
        </p>
      ) : (
        <div className="space-y-4">
          {activeTokens.map((token: ApiTokenResponse) => (
            <TokenPoolRow key={token.id} token={token} />
          ))}
        </div>
      )}
    </div>
  )
}

function TokenPoolRow({ token }: { token: ApiTokenResponse }) {
  const [poolId, setPoolId] = useState(token.defaultPoolId ?? undefined)
  const [saving, setSaving] = useState(false)

  const handleChange = async (newPoolId: string | undefined) => {
    setPoolId(newPoolId)
    setSaving(true)
    try {
      await updateToken({
        data: { id: token.id, defaultPoolId: newPoolId ?? null },
      })
    } finally {
      setSaving(false)
    }
  }

  return (
    <div className="flex items-center gap-4 border rounded-lg p-4">
      <div className="flex-1">
        <p className="font-medium text-sm">{token.name}</p>
        <p className="text-xs text-muted-foreground">
          Last used: {token.lastUsedAt ? new Date(token.lastUsedAt).toLocaleDateString() : "Never"}
        </p>
      </div>
      <div className="w-64">
        <PoolSelector value={poolId} onChange={handleChange} />
      </div>
      {saving && <span className="text-xs text-muted-foreground">Saving...</span>}
    </div>
  )
}
```
