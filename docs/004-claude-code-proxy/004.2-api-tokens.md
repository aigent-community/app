# 004.2 - API Tokens

**Phase 004: Claude Code Proxy** -- Sub-doc 2 of 4: Consumer API token CRUD.

> See also: [004.1 Anthropic Proxy](./004.1-anthropic-proxy.md) | [004.3 Settings UI](./004.3-settings-ui.md) | [004.4 Credit Metering](./004.4-credit-metering.md)

## Overview

CRUD endpoints for consumer API tokens used to authenticate Claude Code requests. Tokens are `ac_` + 32 hex chars, stored as SHA-256 hashes. Each token has an optional `default_pool_id` for pool resolution. Token plaintext shown once on creation.

## Context & Background

Phase 1 defined the `api_token` schema ([001.1](../001-schema-and-auth/001.1-drizzle-schema.md)). Phase 4 implements the full lifecycle: generate, list, update, revoke. These tokens are the auth mechanism for the `/v1/messages` proxy ([004.1](./004.1-anthropic-proxy.md)).

## Goals

- Secure token generation with `crypto.getRandomValues()`
- SHA-256 hash storage (plaintext never persisted)
- CRUD: create, list, update (name/pool), delete (revoke)
- Max 5 active tokens per user
- Handlers + services layer following project conventions

## Non-Goals

- Token scoping beyond `default_pool_id` (no per-model or per-endpoint scopes)
- Token rotation (user deletes old + creates new)
- API key rate limiting (Phase 5)

## 1. Token Generation

File: `apps/data-service/src/hono/services/api-token-service.ts`

### 1.1 Token Format

```
ac_<32 hex chars>
```

Total length: 35 chars. Example: `ac_a1b2c3d4e5f6a7b8c9d0e1f2a3b4c5d6`

### 1.2 Generation + Hashing

```ts
const TOKEN_PREFIX = "ac_"
const TOKEN_BYTE_LENGTH = 16 // 16 bytes = 32 hex chars

export function generateRawToken(): string {
  const bytes = new Uint8Array(TOKEN_BYTE_LENGTH)
  crypto.getRandomValues(bytes)
  const hex = Array.from(bytes)
    .map((b) => b.toString(16).padStart(2, "0"))
    .join("")
  return `${TOKEN_PREFIX}${hex}`
}

export async function hashToken(raw: string): Promise<string> {
  const encoded = new TextEncoder().encode(raw)
  const digest = await crypto.subtle.digest("SHA-256", encoded)
  return Array.from(new Uint8Array(digest))
    .map((b) => b.toString(16).padStart(2, "0"))
    .join("")
}
```

  tokenHash: z.string(),
  name: z.string(),
  defaultPoolId: z.string().nullable(),
  lastUsedAt: z.string().nullable(),

  isActive: z.boolean(),
  createdAt: z.string(),
  updatedAt: z.string(),
})

// ============================================
// Request Schemas
// ============================================

  name: z.string().min(1, "Name required").max(100),
  defaultPoolId: z.string().optional(),
  expiresAt: z.string().datetime().optional(),
})

export const ApiTokenUpdateRequestSchema = z.object({
  name: z.string().min(1).max(100).optional(),
  defaultPoolId: z.string().nullable().optional(),
}).refine(
  (data) => Object.values(data).some((v) => v !== undefined),
  { message: "At least one field required" }

  id: z.string().min(1, "Token ID required"),

export const ApiTokenListResponseSchema = z.object({

## 3. Queries

File: `packages/data-ops/src/queries/api-token.ts`

```ts
import { getDb } from "../database/setup"
import { apiToken } from "../drizzle/schema"
import { eq, and, sql } from "drizzle-orm"

export async function getApiTokenByHash(hash: string) {
  const db = getDb()
  const [row] = await db
    .select()
    .from(apiToken)
    .where(eq(apiToken.tokenHash, hash))
    .limit(1)
  return row ?? null
}

export async function getApiTokensByUserId(userId: string) {
  const db = getDb()
  return db
    .select()
    .from(apiToken)
    .where(eq(apiToken.userId, userId))
    .orderBy(apiToken.createdAt)
}

export async function getActiveTokenCount(userId: string): Promise<number> {
  const db = getDb()
  const [result] = await db
    .select({ count: sql<number>`count(*)` })
    .from(apiToken)
    .where(and(eq(apiToken.userId, userId), eq(apiToken.isActive, true)))
  return result.count
}

export async function createApiToken(
  data: typeof apiToken.$inferInsert
) {
  const db = getDb()
  const [row] = await db.insert(apiToken).values(data).returning()
  return row
}

export async function updateApiToken(
  id: string,
  userId: string,
  data: Partial<Pick<typeof apiToken.$inferSelect, "name" | "defaultPoolId">>
) {
  const db = getDb()
  const [row] = await db
    .update(apiToken)
    .set(data)
    .where(and(eq(apiToken.id, id), eq(apiToken.userId, userId)))
    .returning()
  return row ?? null
}

export async function deleteApiToken(id: string, userId: string) {
  const db = getDb()
  const [row] = await db
    .update(apiToken)
    .set({ isActive: false })
    .where(and(eq(apiToken.id, id), eq(apiToken.userId, userId)))
    .returning()
  return row ?? null
}

export async function updateApiTokenLastUsed(id: string) {
  const db = getDb()
  await db
    .update(apiToken)
    .set({ lastUsedAt: new Date() })
    .where(eq(apiToken.id, id))
}
```

## 4. Service Layer

File: `apps/data-service/src/hono/services/api-token-service.ts`

```ts
import { HTTPException } from "hono/http-exception"
import {
  getApiTokensByUserId,
  getActiveTokenCount,
  createApiToken as createApiTokenQuery,
  updateApiToken as updateApiTokenQuery,
  deleteApiToken as deleteApiTokenQuery,
} from "@repo/data-ops/queries/api-token"
import type {
  ApiTokenCreateInput,
  ApiTokenUpdateInput,
  ApiTokenResponse,
  ApiTokenCreateResponse,
} from "@repo/data-ops/zod-schema/api-token"
import { ulid } from "ulid"

const MAX_ACTIVE_TOKENS = 5

function toResponse(row: Record<string, unknown>): ApiTokenResponse {
  return {
    id: row.id as string,
    userId: row.userId as string,
    name: row.name as string,
    defaultPoolId: row.defaultPoolId as string | null,
    lastUsedAt: row.lastUsedAt ? (row.lastUsedAt as Date).toISOString() : null,
    expiresAt: row.expiresAt ? (row.expiresAt as Date).toISOString() : null,
    isActive: row.isActive as boolean,
    createdAt: (row.createdAt as Date).toISOString(),
    updatedAt: (row.updatedAt as Date).toISOString(),
  }
}

export async function listTokens(userId: string): Promise<ApiTokenResponse[]> {
  const rows = await getApiTokensByUserId(userId)
  return rows.map(toResponse)
}

export async function createToken(
  userId: string,
  data: ApiTokenCreateInput
): Promise<ApiTokenCreateResponse> {
  const activeCount = await getActiveTokenCount(userId)
  if (activeCount >= MAX_ACTIVE_TOKENS) {
    throw new HTTPException(409, {
      message: `Maximum ${MAX_ACTIVE_TOKENS} active tokens allowed`,
    })
  }

  const rawToken = generateRawToken()
  const tokenHash = await hashToken(rawToken)

  const row = await createApiTokenQuery({
    id: ulid(),
    userId,
    tokenHash,
    name: data.name,
    defaultPoolId: data.defaultPoolId ?? null,
    expiresAt: data.expiresAt ? new Date(data.expiresAt) : null,
  })

  return {
    ...toResponse(row),
    token: rawToken, // shown once
  }
}

export async function updateToken(
  userId: string,
  tokenId: string,
  data: ApiTokenUpdateInput
): Promise<ApiTokenResponse> {
  const updateData: Record<string, unknown> = {}
  if (data.name !== undefined) updateData.name = data.name
  if (data.defaultPoolId !== undefined) updateData.defaultPoolId = data.defaultPoolId

  const row = await updateApiTokenQuery(tokenId, userId, updateData)
  if (!row) throw new HTTPException(404, { message: "Token not found" })
  return toResponse(row)
}

export async function revokeToken(
  userId: string,
  tokenId: string
): Promise<void> {
  const row = await deleteApiTokenQuery(tokenId, userId)
  if (!row) throw new HTTPException(404, { message: "Token not found" })
}
```

## 5. Route Handlers

File: `apps/data-service/src/hono/handlers/token-handlers.ts`

```ts
import { Hono } from "hono"
import { zValidator } from "@hono/zod-validator"
import {
  ApiTokenCreateRequestSchema,
  ApiTokenUpdateRequestSchema,
  ApiTokenIdParamSchema,
} from "@repo/data-ops/zod-schema/api-token"
import { sessionAuth } from "../middleware/auth"
import * as tokenService from "../services/api-token-service"

const tokens = new Hono<{ Bindings: Env }>()

tokens.use("/*", sessionAuth())

// GET /api/tokens
tokens.get("/", async (c) => {
  const userId = c.get("userId")
  const data = await tokenService.listTokens(userId)
  return c.json({ data })
})

// POST /api/tokens
tokens.post(
  "/",
  zValidator("json", ApiTokenCreateRequestSchema),
  async (c) => {
    const userId = c.get("userId")
    const data = c.req.valid("json")
    return c.json(await tokenService.createToken(userId, data), 201)
  }
)

// PATCH /api/tokens/:id
tokens.patch(
  "/:id",
  zValidator("param", ApiTokenIdParamSchema),
  zValidator("json", ApiTokenUpdateRequestSchema),
  async (c) => {
    const userId = c.get("userId")
    const { id } = c.req.valid("param")
    const data = c.req.valid("json")
    return c.json(await tokenService.updateToken(userId, id, data))
  }
)

// DELETE /api/tokens/:id
tokens.delete(
  "/:id",
  zValidator("param", ApiTokenIdParamSchema),
  async (c) => {
    const userId = c.get("userId")
    const { id } = c.req.valid("param")
    await tokenService.revokeToken(userId, id)
    return c.body(null, 204)
  }
)

export default tokens
```

| Method | Path | Auth | Request Body | Response | Status |
|--------|------|------|-------------|----------|--------|
| GET | `/api/tokens` | session | - | `ApiTokenListResponse` | 200 |
| POST | `/api/tokens` | session | `ApiTokenCreateRequest` | `ApiTokenCreateResponse` | 201 / 409 |
| PATCH | `/api/tokens/:id` | session | `ApiTokenUpdateRequest` | `ApiTokenResponse` | 200 / 404 |
| DELETE | `/api/tokens/:id` | session | - | - | 204 / 404 |

## 8. Error Cases

| Scenario | Status | Body |
|----------|--------|------|
| No session | 401 | `{ "error": "Authentication required" }` |
| Max 5 active tokens exceeded | 409 | `{ "error": "Maximum 5 active tokens allowed" }` |
| PATCH/DELETE token not found or not owned | 404 | `{ "error": "Token not found" }` |
| Zod validation failure | 400 | `{ "error": "Validation failed", ... }` |

## Verification Checklist

- [ ] `POST /api/tokens` generates `ac_` + 32 hex chars token
- [ ] Token stored as SHA-256 hash (plaintext not persisted)
- [ ] Plaintext token returned in creation response only
- [ ] `GET /api/tokens` returns user's tokens (hash excluded)
- [ ] `PATCH /api/tokens/:id` updates name and/or `default_pool_id`
- [ ] `DELETE /api/tokens/:id` soft-deletes (sets `isActive: false`)
- [ ] Max 5 active tokens enforced (409 on exceed)
- [ ] Token ownership enforced (userId match on all mutations)
- [ ] `last_used_at` updated on each proxy request (fire-and-forget)
